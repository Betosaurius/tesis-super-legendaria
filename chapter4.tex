\chapter{\texorpdfstring{$(P_5,\text{house},C_5)$}{(P5,house,C5)}-free graphs}
\label{cap:no_c5}

\section{Characterization of monopolarity}
\label{sec:no_c5_characterization}

Thus far, \textsc{Monopolarity} has been solved for those
$\p{P_5,\text{house}}$-free graphs that are either chordal or have an induced
copy of $C_5$. If a graph in this family is neither chordal nor it has an
induced copy of $C_5$, it follows that it must have an induced copy of $C_4$.
Thus, all graphs considered in this section, unless otherwise stated, are
assumed to have an induced copy of $C_4$, which will be denoted by
$C=\p{a,b,c,d,a}$, and $Z=\l{a,b,c,d}$.

It turns out that there is a single additional minimal obstruction to
monopolarity for $C_5$-free graphs.

Since all graphs in the current family have an induced $C_4$, we will analyze
the exact neighborhoods of an induced copy of $C_4$. The next lemma
straightforwardly tells us that some bags must be empty for the graph to be
$(P_5,\text{house})$-free.

\begin{lemma}
    \label{lem:pre_neighborhoods_no_c5}
    Let $G$ be a connected $(P_5,\text{house},C_5)$-free graph with an induced
    copy of $C_4$ having $Z$ as vertex set. Let $X$ be a subset of $Z$.
    \begin{lemenum}
        \item If $N_X$ is a 2-bag adjacent to consecutive vertices in $Z$, then
            $N_X$ is empty. \label{lem:type_1}
        \item The set $N(x)$ is $C_4$-free for every $x\in V$. In particular,
            $N_Z=\varnothing$. \label{lem:type_2}
    \end{lemenum}
\end{lemma}

\begin{proof}
    \leavevmode \par
    \begin{enumerate}[label=(\roman*)]
        \item Suppose without loss of generality that $X=\l{a,b}$. If $x\in N_X$,
            then the set $\l{a,b,c,d,x}$ induces a copy of the house (see
            \Cref{fig:type_1}). Therefore, $N_X$ is empty.

            \begin{figure}[htb!]
                \centering
                \begin{tikzpicture}[every node/.style={vertex}, node
                    distance=1cm, on grid, edge]
                    \foreach \i in {0,...,3}
                        \node (\i)  at ({(360/4)*\i + 45}:1){};
                    
                    \foreach \i in {0,...,3}{
                        \draw let \n1={int(mod(\i+1,4))} in (\i) to (\n1);
                    }

                    \node (r) at (0,1.5){};
                    \draw (r) to (0);
                    \draw (r) to (1);
                \end{tikzpicture}
                \caption{An induced copy of the house given that $N_X$ is nonempty.}
                \label{fig:type_1}
            \end{figure}
        \item If $x\in V$ is such that its neighborhood has an induced copy of
            $C_4$, then such a copy, along with $x$, induces a copy of $H_{11}$
            (see \Cref{fig:type_2}). Thus, $N(x)$ is $C_4$-free, and
            consequently, $N_Z=\varnothing$.

                    $C_4$-free.}
                \label{fig:type_2}
            \end{figure}
    \end{enumerate}
\end{proof}

From \Cref{lem:pre_neighborhoods_no_c5} follows that there is only one type of
2-bag that can be nonempty. Additionally, is clear that there is only one type
of 3-bag and only one type of 1-bag. Thus, for the following lemmas, we prove
the claims only for the bags that have a chance to be nonempty, since for all
other the claims are vacuously true.

\begin{lemma}
    Let $G$ be a connected $(P_5,\text{house},C_5)$-free graph with and induced
    copy of $C_4$ having vertex set $Z$ and free of the graphs depicted in
    Figures \ref{fig:chordal-p5-free-obstructions},
    \ref{fig:p5-house-obstruction-with-c5} and
    \ref{fig:p5-house-obstruction-no-c5}. Let $X$ and $Y$ be different subsets
    of $Z$.
    \begin{lemenum}
        \item If $N_X$ is a 3-bag and $N_Y$ is a bag such that $|X\cap Y|=2$ and
            $Y$ does not contain the middle vertex of $X$, then $N_X$ is
            anticomplete to $N_Y$. \label{lem:anti_1}
        \item If $N_X$ is a 2-bag and $N_Y$ is a 1-bag such that $|X\cap Y|=1$,
            then $N_X$ is anticomplete to $N_Y$. \label{lem:anti_2}
        \item If $N_X$ is a 3-bag and $N_Y$ is a 1-bag nonadjacent to the middle
            vertex of $X$, then $N_X$ is anticomplete to $N_Y$.
            \label{lem:anti_3}
        \item If $N_X$ and $N_Y$ are 1-bags adjacent to consecutive vertices in
            $Z$, then $N_X$ is anticomplete to $N_Y$. \label{lem:anti_4}
        \item If $N_X$ is a 1-bag, then $N_\varnothing$ is anticomplete to
            $N_X$. \label{lem:anti_5}
    \end{lemenum} 
\end{lemma}

            \Cref{fig:anti_1}). Then, $N_X$ is anticomplete to $N_Y$.

\begin{lemma}
    Let $G$ be a connected $(P_5,\text{house},C_5)$-free graph with and induced
    copy of $C_4$ having $Z$ as vertex set and free of the graphs depicted in
    Figures \ref{fig:chordal-p5-free-obstructions},
    \ref{fig:p5-house-obstruction-with-c5} and
    \ref{fig:p5-house-obstruction-no-c5}. Let $X$ and $Y$ be different subsets
    of $Z$.
    \begin{lemenum}
        \item If $N_X$ is a 3-bag, then $N_X$ is a clique.
            \label{lem:type_3}
        \item If $N_X$ is a 2-bag, then $G[N_X]$ is a cluster.
            \label{lem:type_4}
        \item If $N_X$ and $N_Y$ are 3-bags with adjacent middle vertices, then
            one between $N_X$ and $N_Y$ is empty. \label{lem:type_5}
        \item If both $N_X$ and $N_Y$ are 1-bags adjacent to non-consecutive
            vertices in $Z$, then one between $N_X$ and $N_Y$ is empty.
            \label{lem:type_6}
        \item If $N_X$ is a 1-bag, then $G[N_X]$ is either a cluster or split.
            \label{lem:type_7}
        \item For any nonempty $X$, every connected component of
            $G[N_\varnothing]$ is homogeneous to $N_X$. Moreover, each connected
            component of $G[N_\varnothing]$ is anticomplete to all bags but one.
            \label{lem:type_8}
        \item Let $N_X$ and $N_Y$ be 3-bags, and let $N_W$ be a 2-bag such that
            $W$ does no contain the middle vertices of $X$ and $Y$. Then,
            $N_\varnothing$ is adjacent to at most one of $N_X$, $N_Y$ and
            $N_W$. \label{lem:type_9}
        \item If $N_X$ is a 3-bag, then non-isolated vertices in $N_\varnothing$
            are anticomplete to $N_X$. \label{lem:type_10}
        \item If $N_X$ is a 3-bag and $N_Y$ is the 1-bag adjacent to the middle
            vertex of $X$, then non-isolated vertices in $N_Y$ are complete to
            $N_X$.
            \label{lem:type_11}
        \item If $N_X$ and $N_Y$ are 1-bags adjacent to consecutive vertices in
            $Z$, then at least one between $G[N_X]$ and $G[N_Y]$ is a cluster.
            \label{lem:type_12}
        \item If $N_X$ and $N_Y$ are 1-bags adjacent to consecutive vertices in
            $Z$, then at least one between $G[N_X]$ and $G[N_Y]$ is split.
            \label{lem:type_13}
    \end{lemenum}
\end{lemma}

        \item The set $N_{bd}$ is stable. \label{lem:type_3bag_1}
        \item The graphs $G[N_a]$ and $G[N_c]$ are clusters.
            \label{lem:type_3bag_2}

\begin{theorem}
    A connected $(P_5,\text{house},C_5)$-free graph $G$ with an induced copy of
    $C_4$ is monopolar if and only if it does not contain any of graphs depicted
    in Figures \ref{fig:chordal-p5-free-obstructions},
    \ref{fig:p5-house-obstruction-with-c5} and
    \ref{fig:p5-house-obstruction-no-c5}, as an induced subgraph.
\end{theorem}

\begin{proof}
    Since monopolarity is hereditary, it follows from
    \Cref{prop:minimal_obs_no_c5,prop:minimal_obs_c5,prop:minimal_obs_chordal}
    that $G$ does not contain the mentioned graphs as induced subgraphs.

    Due to \Cref{lem:type_1,lem:type_2}, the only possibly nonempty exact
    neighborhoods of the $4$-cycle $C$ are
    \[
        N_\varnothing, N_{a}, N_{b}, N_{c}, N_{d}, N_{ac}, N_{bd}, N_{abc}, N_{bcd},
        N_{acd} \text{ and } N_{abd}.
    \]
    We proceed by considering the following cases:

    \paragraph{Case 1: At least one 3-bag is nonempty.} Suppose without loss of
    generality that $N_{abc}$ is nonempty and let $b'\in N_{abc}$. By
    \Cref{lem:type_5}, both $N_{bcd}$ and $N_{abd}$ are empty, so the only other
    possibly nonempty 3-bag is $N_{acd}$. Furthermore, by \Cref{lem:type_6},
    suppose without loss of generality that $N_c$ is empty. Additionally,
    \Cref{lem:anti_1} ensures that $N_{abc}$ is anticomplete to $N_{acd}$, while
    \Cref{lem:type_3} implies that both sets are cliques. Therefore,
    \[
        N_{abc} \cup N_{acd} \cup \l{b,d}
    \]
    is a 2-cluster. Moreover, \Cref{lem:type_3bag_2} implies that $G[N_a]$ is a
    cluster, and \Cref{lem:anti_3} ensures that $N_a$ is anticomplete to both
    $N_{abc}$ and $N_{acd}$, so
    \[
        N_{abc}\cup N_{acd} \cup \l{b,d} \cup N_a
    \]
    is a cluster.

    On the one hand, from \Cref{lem:type_3bag_1}, we have that $N_{bd}$ is
    stable, and from \Cref{lem:comp_1}, we have that $N_{bd}$ is complete to
    both $N_{abc}$ and $N_{acd}$. On the other hand, by \Cref{lem:anti_2},
    $N_{ac}$ is anticomplete to $N_a$, while \Cref{lem:type_4} implies that
    $G[N_{ac}]$ is a cluster. Also, $N_{abc}$ and $N_{acd}$ are anticomplete to
    $N_{ac}$ by \Cref{lem:anti_1}. Then, up to this point,
    \[
        \p{N_{bd} \cup \l{a,c} , N_{abc} \cup N_{acd} \cup N_{ac} \cup N_{a} 
        \cup \l{b,d}}
    \]
    is a partial monopolar partition of $G$.

    For $N_\varnothing$, let $E_{bd}$ be the set of vertices in
    $N_{\varnothing}$ that are adjacent to $N_{bd}$. By \Cref{lem:type_3bag_3},
    $E_{bd}$ induces a cluster, while \Cref{lem:type_8} implies that such a set
    is anticomplete to all other bags. Therefore,
    \[
        N_{abc} \cup N_{acd} \cup N_{ac} \cup N_{a} 
        \cup \l{b,d} \cup E_{bd}
    \]
    is a cluster. For vertices in $N_\varnothing\setminus E_{bd}$, by
    \Cref{lem:type_9}, they are adjacent to just one bag between $N_{abc}$,
    $N_{acd}$ and $N_{ac}$, and \Cref{lem:type_8} implies that
    $N_\varnothing\setminus E_{bd}$ is anticomplete to $E_{bd}$. Consider the
    following subcases:

    \paragraph{Case 1.1: $N_\varnothing\setminus E_{bd}$ is adjacent to
    $N_{ac}$.} Let $S$ be the set of vertices in trivial components of
    $G[N_\varnothing\setminus E_{bd}]$. Clearly,
    \[
        N_{bd} \cup \l{a,c} \cup S
    \]
    is a stable set. Meanwhile, let $\mathcal{C}$ be the set of connected
    components of $G[N_\varnothing\setminus E_{bd}\setminus S]$. All such
    components are nontrivial. Let $C\in \mathcal{C}$. By
    \Cref{lem:type_3bag_4}, $C$ is split and there is a unique connected
    component of $G[N_{ac}]$, $C'$, to which $C$ is adjacent, and the same item
    ensures that $C'$ is anticomplete to any other $D\in \mathcal{C}$. Recall
    that $G[N_{ac}]$ is a cluster, so $C'$ is a clique. Then, if $(S_C,K_C)$ is
    a split partition of $C$, we have that $K_C\cup C'$ is a clique anticomplete
    to all other connected components of $G[N_{ac}]$ and $G[E_{ac}]$. Thus,
    \[
        N_{abc} \cup N_{acd} \cup N_{ac} \cup N_{a} 
        \cup \l{b,d} \cup E_{bd} \cup \bigcup_{C\in \mathcal{C}} K_C
    \]
    is a cluster, while
    \[
        N_{bd} \cup \l{a,c} \cup S \cup \bigcup_{C\in \mathcal{C}} S_C
    \]
    is a stable set.
    
    Finally, \Cref{lem:type_6} ensures that one between $N_{b}$ and $N_{d}$ is
    empty. Take the following subcases:

    \paragraph{Case 1.1.1: $N_d$ is empty.} Due to \Cref{lem:anti_2}, $N_b$ is
    anticomplete to $N_{bd}$. Let $K(N_b,N_{abc})$ be the subset of vertices in
    $N_b$ that are complete to $N_{abc}$, and let $D = N_b \setminus
    K(N_b,N_{abc})$. By \Cref{lem:type_11}, $D$ is stable, so
    \[
        N_{bd} \cup \l{a,c} \cup S \cup \bigcup_{C\in\mathcal{C}} S_C \cup D
    \]
    is a stable set. Additionally, \Cref{lem:type_11} implies that
    $K(N_b,N_{abc})$ is anticomplete to $D$.
    
    Let $F$ be the set of vertices in $K(N_b,N_{abc})$ that are adjacent to the
    set $N_{ac}$. By \Cref{lem:type_3bag_6}, $F$ is stable, and the same lemma
    implies that $F$ is anticomplete to $K(N_b,N_{abc})\setminus F$. Since
    $K(N_b,N_{abc})$ is anticomplete to $D$, $F$ is anticomplete to $D$, so
    \[
        N_{bd} \cup \l{a,c} \cup S \cup \bigcup_{C\in\mathcal{C}} S_C \cup D 
        \cup F
    \]
    is a stable set. Finally, notice that $G[K(N_b,N_{abc})\setminus F]$ is
    split by \Cref{lem:type_3bag_51}. Then, there is a partition of
    $K(N_b,N_{abc})\setminus F$ into $S_b$ a stable set and $K_b$ a clique.
    Since $K(N_b,N_{abc})$ is anticomplete to $D$ and $K(N_b,N_{abc})\setminus
    F$ is anticomplete to $F$, it follows that $S_b$ is anticomplete to both $D$
    and $F$. Meanwhile, we already know that $K_b$ is anticomplete to $N_{abc}$
    and $N_{ac}$, and it is anticomplete to $N_a$ and $N_{acd}$ due to
    \Cref{lem:anti_4,lem:anti_3}, respectively. It follows that
    \begin{align*}
        \bigg( 
            & N_{bd} \cup \l{a,c} \cup S \cup \bigcup_{C\in\mathcal{C}} S_C \cup D 
            \cup F \cup S_b , \\
            & N_{abc} \cup N_{acd} \cup N_{ac} \cup N_{a} \cup \l{b,d} \cup E_{bd} 
            \cup \bigcup_{C\in \mathcal{C}} K_C \cup K_b \bigg)
    \end{align*}
    is a monopolar partition of $G$.

    \paragraph{Case 1.1.2: $N_b$ is empty.} Notice that we can suppose that
    $N_{acd}$ is empty, for if not, then we proceed as in the previous subcase.
    By \Cref{lem:type_3bag_5}, $G[N_d]$ is split, so there is a split partition
    $(S_d,K_d)$ with $S_d$ a maximal stable set. By \Cref{lem:anti_2}, $N_d$ is
    anticomplete to $N_{bd}$, while \Cref{lem:anti_3} implies that $N_d$ is
    anticomplete to $N_{abc}$. Finally, from \Cref{lem:type_3bag_7} and the
    maximality of $S_d$, we have that $K_d$ is anticomplete to $N_{ac}$, so
   \begin{align*}
        \bigg( 
            & N_{bd} \cup \l{a,c} \cup S \cup \bigcup_{C\in\mathcal{C}} S_C 
            \cup S_d , \\
            & N_{abc} \cup N_{ac} \cup N_{a} \cup \l{b,d} \cup E_{bd} 
            \cup \bigcup_{C\in \mathcal{C}} K_C \cup K_d \bigg)
    \end{align*}
    is a monopolar partition of $G$.

    \paragraph{Case 1.2: $N_\varnothing\setminus E_{bd}$ is adjacent to
    $N_{abc}\cup N_{acd}$.} Suppose without loss of generality that
    $N_\varnothing \setminus E_{bd}$ is adjacent to $N_{abc}$ so it is
    anticomplete to $N_{acd}$. By \Cref{lem:type_10}, $N_\varnothing\setminus
    E_{bd}$ is stable, so
    \[
        N_{bd} \cup \l{a,c} \cup N_\varnothing\setminus E_{bd}
    \]
    is stable. 
    
    We claim that $N_d$ is empty. Indeed, let $w\in N_\varnothing\setminus
    E_{bd}$ be adjacent to $x\in N_{abc}$. Recall that $N_d$ is anticomplete to
    $N_{abc}$ and $N_\varnothing$ by \Cref{lem:anti_3,lem:anti_5}, respectively.
    Then, if $d'\in N_d$, we have that $\p{w,x,a,d,d'}$ is an induced copy of
    $P_5$ (see \Cref{fig:Nd_empty}).

    Additionally, the analysis for $N_b$ in Case 1.1.1 did not depend on which
    set $N_\varnothing \setminus E_{bd}$ was adjacent to. Since
    \Cref{lem:anti_5} ensures that $N_\varnothing$ is anticomplete to $N_b$, we
    have that
    \begin{align*}
        \bigg( 
            & N_{bd} \cup \l{a,c} \cup N_\varnothing\setminus E_{bd} \cup D 
            \cup F \cup S_b , \\
            & N_{abc} \cup N_{acd} \cup N_{ac} \cup N_{a} \cup \l{b,d} \cup E_{bd} 
            \cup K_b \bigg)
    \end{align*}
    is a monopolar partition of $G$.

    \paragraph{Case 2: All 3-bags are empty.} Suppose $u$ and $v$ are adjacent
    vertices in $N_{ac}$. Then, $\p{a,u,c,d,a}$ is a $4$-cycle with a nonempty
    $3$-bag, since $v\in N_{acu}$, so one could proceed as in the previous case
    (see \Cref{fig:3_bags_empty}).

    Thus, consider that $N_{ac}$ is stable. Similarly, consider $N_{bd}$ to be
    stable. As in the previous case, by \Cref{lem:type_6}, suppose without loss
    of generality that both $N_{c}$ and $N_{d}$ are empty.

    By \Cref{lem:type_12}, at least one between $G[N_a]$ and $G[N_b]$ is a
    cluster; suppose without loss of generality that $G[N_b]$ is a cluster.
    Consider the following subcases: \todo{here}

    \paragraph{Case 2.1: $G[N_a]$ is split.} Let $(S,K)$ be a partition of $N_a$
    with $S$ a stable set and $K$ a maximal clique. By \Cref{lem:anti_2}, $N_a$
    is anticomplete to $N_{ac}$ and $N_b$ is anticomplete to $N_{bd}$. Then, $S$
    is anticomplete to $N_{ac}$. If $K$ is nonempty, due to its maximality, it
    has at least two vertices, $u$ and $v$. If just one of them is adjacent to
    $w\in N_{bd}$, say $u$, then $\l{u,v,a,b,w}$ induce a house. On the
    contrary, if both $u$ and $v$ are adjacent to $w$, then $\p{a,b,w,u,a}$ is a
    $4$-cycle with a nonempty $3$-bag, since $v\in N_{avw}$. Thus, it follows
    that $K$ is anticomplete to $N_{bd}$. At any rate,
    \[
        \p{N_{ac} \cup \l{b,d} \cup S, N_{bd} \cup N_b \cup \l{a,c} \cup K}
    \]
    is a partial monopolar partition of $G$, with $N_\varnothing$ still to be
    assessed. If $N_{\varnothing}$ is empty, then we are done. If not, by
    \Cref{lem:type_8}, each connected component of $N_\varnothing$ is not
    anticomplete to one, and only one, between $N_{ac}$ and $N_{bd}$. Suppose
    without loss of generality that $N_{\varnothing}$ is not anticomplete to
    $N_{ac}$, and let $D$ be the set of vertices in $N_\varnothing$ that are not
    anticomplete to $N_{ac}$. Let $u$ and $v$ be vertices in $N_{ac}$. Suppose
    there are nonadjacent vertices $u'$ and $v'$ in $D$ such that $u$ is
    adjacent to $u'$ and nonadjacent to $v'$, and $v$ is adjacent to $v'$ and
    nonadjacent to $u'$. Then, since $N_{ac}$ is stable, $\p{u,u',b,v,v'}$ is a
    $P_5$. This implies that the vertices in $N_{ac}$ are ordered by the
    connected components to which they are adjacent.
    
    Let $x$ be the vertex in $N_{ac}$ that is adjacent to all connected
    components of $D$. Notice that $x$ is complete to all such components, so
    $x$ is complete to $D$. Observe that $\p{a,b,c,x,a}$ is a $C_4$ with the set
    $D$ complete to $x$ and anticomplete to all other vertices of the $C_4$.
    Then, by \Cref{lem:type_7}, $G[D]$ is either a cluster or split. Let $u$ be
    in $D$. If $v$ and $w$ are adjacent vertices in $N_b$, then $x$ is adjacent
    to both, otherwise, $\p{u,x,c,b,v}$ or $\p{u,x,c,b,w}$ are a $P_5$. Then,
    $\p{a,b,v,x,a}$ is a $C_4$ with a nonempty 3-bag, since $w\in N_{bvx}$.
    Thus, $N_b$ is stable. Consider the following subcases:

    \paragraph{Case 2.1.1: $E = N_\varnothing\setminus D$ is empty.} Then $D =
    N_\varnothing$. If $G[D]$ is a cluster, then
    \[
        \p{N_{ac} \cup \l{b,d} \cup S, N_{bd} \cup N_b \cup \l{a,c} \cup K \cup D}
    \]
    is a monopolar partition of $G$. On the contrary, if $G[D]$ is not a
    cluster, then it is split. Additionally, due to $H_{7}$, $G[N_a]$ is a
    cluster. Let $(S',K')$ be a partition of $D$ with $S'$ a stable set and $K'$
    a maximal clique. If $K'$ is not trivial, then $x$ is the only vertex in
    $N_{ac}$ that is not anticomplete to $K'$, otherwise, one would get a $C_4$
    with a nonempty $3$-bag. Therefore,
    \[
        \p{N_{bd} \cup N_b \cup \l{a,c} \cup S' , N_{ac} \cup N_a \cup \l{b,d} 
        \cup K'}
    \]
    is a monopolar partition of $G$.

    \paragraph{Case 2.1.2: $E$ is nonempty.} Similar to the case with $D$, there
    is a vertex $y\in N_{bd}$ complete to $E$, $N_a$ is stable, and $G[E]$ is
    either a cluster or split. Moreover, due to $P_5$, $x$ is adjacent to $y$.
    Then, due to $H_{7}$ at most one between $G[D]$ and $G[E]$ is not a cluster.
    Similarly, due to $H_{5}$, at most one between $G[D]$ and $G[E]$ is not
    split. Suppose without loss of generality that $G[D]$ is split.

    If $G[E]$ is a cluster, then, as in the last partition, let $(S',K')$ be a
    split partition of $G[D]$ with $K'$ a maximal clique. Hence,
    \[
        \p{N_{bd} \cup N_b \cup \l{a,c} \cup S' , N_{ac} \cup N_a \cup \l{b,d} 
        \cup K' \cup E}
    \]
    is a monopolar partition of $G$. On the contrary, if $G[E]$ is not a
    cluster, then $G[D]$ is a cluster and $G[E]$ is split. Then, if $(S',K')$ is
    a split partition of $E$ with $K'$ a maximal clique, it follows that
    \[
        \p{N_{ac} \cup N_a \cup \l{b,d} \cup S' , N_{bd} \cup N_b \cup \l{a,c} 
        \cup K' \cup D}
    \]
    is a monopolar partition of $G$.

    \paragraph{Case 2.2: $G[N_a]$ is not split.} Then, by \Cref{lem:type_13},
    $G[N_b]$ is split. In particular, due to \Cref{lem:type_7}, it must be that
    $G[N_a]$ is a cluster. Then, $N_a$ and $N_b$ are swapped as in the previous
    case, so
    \[
        \p{N_{bd} \cup N_b \cup \l{a,c} \cup S , N_{ac} \cup \l{b,d} \cup K}
    \]
    is a partial monopolar partition of $G$, with $(S,K)$ a split partition of
    $N_b$ with $S$ a stable set and $K$ a maximal clique. All cases of
    $N_\varnothing$ explained in the previous subcase still give monopolar
    partitions just by swapping the stable set with the set inducing the
    cluster.
\end{proof}
