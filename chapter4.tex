\chapter{\texorpdfstring{$(P_5,\text{house},C_5)$}{(P5,house,C5)}-free graphs}
\label{cap:no_c5}

Thus far, \textsc{Monopolarity} has been solved for those
$\p{P_5,\text{house}}$-free graphs that are either chordal or have an induced
copy of $C_5$. If a graph in this family is neither chordal nor it has an
induced copy of $C_5$, it follows that it must have an induced copy of $C_4$.
Thus, all graphs considered in this section, unless otherwise stated, are
assumed to have an induced copy of $C_4$, which will be denoted by
$C=\p{a,b,c,d,a}$, and $Z=\l{a,b,c,d}$.

\section{Characterization of monopolarity}
\label{sec:no_c5_characterization}

It turns out that there is a single additional minimal obstruction to
monopolarity for $C_5$-free graphs.

Since all graphs in the current family have an induced $C_4$, we analyze the
exact neighborhoods of an induced copy of $C_4$. The next lemma
straightforwardly tells us that some bags must be empty for the graph to be
$(P_5,\text{house})$-free.

\begin{lemma}
    \label{lem:pre_neighborhoods_no_c5}
    Let $G$ be a connected $(P_5,\text{house},C_5)$-free graph with an induced
    copy of $C_4$ having $Z$ as vertex set. Let $X$ be a subset of $Z$.
    \begin{lemenum}
        \item If $N_X$ is a 2-bag complete to consecutive vertices in $Z$, then
            $N_X$ is empty. \label{lem:type_1}
        \item The set $N(x)$ is $C_4$-free for every $x\in V$. In particular,
            $N_Z=\varnothing$. \label{lem:type_2}
    \end{lemenum}
\end{lemma}

\begin{proof}
    \leavevmode \par
    \begin{enumerate}[label=(\roman*)]
        \item Suppose without loss of generality that $X=\l{a,b}$. If $x\in N_X$,
            then the set $\l{a,b,c,d,x}$ induces a copy of the house (see
            \Cref{fig:type_1}). Therefore, $N_X$ is empty.

            \begin{figure}[htb!]
                \centering

                    distance=1cm, on grid, edge]

                    }

                    \draw (r) to (0);
                    \draw (r) to (1);
                \end{tikzpicture}
                \caption{An induced copy of the house given that $N_X$ is nonempty.}
                \label{fig:type_1}
            \end{figure}
        \item If $x\in V$ is such that its neighborhood has an induced copy of
            $C_4$, then such a copy, along with $x$, induces a copy of $H_{11}$
            (see \Cref{fig:type_2}). Thus, $N(x)$ is $C_4$-free, and
            consequently, $N_Z=\varnothing$.

From \Cref{lem:pre_neighborhoods_no_c5} it follows that there is only one type
of 2-bag that can be nonempty. Additionally, is clear that there is only one
type of 3-bag and only one type of 1-bag. Thus, for the following lemmas, we
prove the claims only for the bags that have a chance to be nonempty, since for
all others the claims are vacuously true. The next lemma asserts relations of
anticompleteness between sets.

\begin{lemma}
    Let $G$ be a connected $(P_5,\text{house},C_5)$-free graph with an induced
    copy of $C_4$ having vertex set $Z$ and free of the graphs depicted in
    Figures \ref{fig:chordal-p5-free-obstructions},
    \ref{fig:p5-house-obstruction-with-c5} and
    \ref{fig:p5-house-obstruction-no-c5}. Let $X$ and $Y$ be different subsets
    of $Z$.
    \begin{lemenum}
        \item If $N_X$ is a 3-bag and $N_Y$ is a bag anticomplete to the middle
        vertex of $X$ and such that $|X\cap Y|=2$, then $N_X$ is anticomplete to
        $N_Y$. \label{lem:anti_1}
        \item If $N_X$ is a 2-bag and $N_Y$ is a 1-bag such that $|X\cap Y|=1$,
            then $N_X$ is anticomplete to $N_Y$. \label{lem:anti_2}
        \item If $N_X$ is a 3-bag and $N_Y$ is a 1-bag anticomplete to the
            middle vertex of $X$, then $N_X$ is anticomplete to $N_Y$.
            \label{lem:anti_3}
        \item If $N_X$ and $N_Y$ are 1-bags complete to consecutive vertices in
            $Z$, then $N_X$ is anticomplete to $N_Y$. \label{lem:anti_4}
        \item If $N_X$ is a 1-bag, then $N_\varnothing$ is anticomplete to
            $N_X$. \label{lem:anti_5}
    \end{lemenum} 
\end{lemma}

The following claims are regarding completeness relations among sets.

\begin{lemma}
    Let $G$ be a connected $(P_5,\text{house},C_5)$-free graph with an induced
    copy of $C_4$ having $Z$ as vertex set and free of the graphs depicted in
    Figures \ref{fig:chordal-p5-free-obstructions},
    \ref{fig:p5-house-obstruction-with-c5} and
    \ref{fig:p5-house-obstruction-no-c5}. Let $X$ and $Y$ be different subsets
    of $Z$. 
    \begin{lemenum}
        \item If $N_X$ is a 3-bag and $N_Y$ is a 2-bag complete to the middle
            vertex of $X$, then $N_X$ is complete to $N_Y$. \label{lem:comp_1}
        \item If $N_X$ is a 3-bag and $N_Y$ is the 1-bag complete to the middle
            vertex of $X$, then non-isolated vertices in $N_Y$ are complete to
            $N_X$. \label{lem:type_11}
        \item Let $E_X$ be the set of vertices in $N_\varnothing$ not
            anticomplete to $N_X$ and let $N_Y$ be a 1-bag such that $X\cap Y =
            \varnothing$. If $x$ is a vertex in $N_X$ not anticomplete to $E_X$,
            then $x$ is complete to $N_Y$. \label{lem:type_15}
    \end{lemenum}

\end{lemma}

\begin{proof}
    \leavevmode \par
    \begin{enumerate}[label=(\roman*)]
        \item Suppose without loss of generality that $X=\l{a,b,c}$ and
            $Y=\l{b,d}$. If $x\in N_X$ is not adjacent to $y\in N_Y$, then
            $\l{a,b,d,x,y}$ induces a copy of the house (see \Cref{fig:comp_1}).
            Hence, $N_X$ is complete to $N_Y$.

\begin{lemma}
    Let $G$ be a connected $(P_5,\text{house},C_5)$-free graph with an induced
    copy of $C_4$ having $Z$ as vertex set and free of the graphs depicted in
    Figures \ref{fig:chordal-p5-free-obstructions},
    \ref{fig:p5-house-obstruction-with-c5} and
    \ref{fig:p5-house-obstruction-no-c5}. Let $X$ and $Y$ be different subsets
    of $Z$.
    \begin{lemenum}
        \item If $N_X$ is a 3-bag, then $N_X$ is a clique.
            \label{lem:type_3}
        \item If $N_X$ is a 2-bag, then $G[N_X]$ is a cluster.
            \label{lem:type_4}
        \item If $N_X$ and $N_Y$ are 3-bags with adjacent middle vertices, then
            at least one between $N_X$ and $N_Y$ is empty. \label{lem:type_5}
        \item If both $N_X$ and $N_Y$ are 1-bags complete to non-consecutive
            vertices in $Z$, then at least one between $N_X$ and $N_Y$ is empty.
            \label{lem:type_6}
        \item If $N_X$ is a 1-bag, then $G[N_X]$ is either a cluster or split.
            \label{lem:type_7}
        \item If $N_X$ and $N_Y$ are 1-bags complete to consecutive vertices in
            $Z$, then at least one between $G[N_X]$ and $G[N_Y]$ is a cluster.
            \label{lem:type_12}
        \item If $N_X$ and $N_Y$ are 1-bags complete to consecutive vertices in
            $Z$, then at least one between $G[N_X]$ and $G[N_Y]$ is split.
            \label{lem:type_13}
    \end{lemenum}
\end{lemma}

there are still some useful facts.

\begin{lemma}

    copy of $C_4$ having $Z$ as vertex set and free of the graphs depicted in
    Figures \ref{fig:chordal-p5-free-obstructions},
    \ref{fig:p5-house-obstruction-with-c5} and
    \ref{fig:p5-house-obstruction-no-c5}. Let $X$ and $Y$ be different subsets
    of $Z$.
    \begin{lemenum}
        \item For any nonempty $X$, every connected component of
            $G[N_\varnothing]$ is homogeneous to $N_X$. Moreover, each connected
            component of $G[N_\varnothing]$ is anticomplete to all bags but one.
            \label{lem:type_8}
        \item Let $N_X$ and $N_Y$ be 3-bags, and let $N_W$ be a 2-bag
            anticomplete to the middle vertices of $X$ and $Y$. Then,
            $N_\varnothing$ is anticomplete to all but at most one bag between
            $N_X$, $N_Y$ and $N_W$. \label{lem:type_9}
        \item If $N_X$ is a 3-bag, then non-isolated vertices in $N_\varnothing$
            are anticomplete to $N_X$. \label{lem:type_10}

\end{lemma}

If all 3-bags are empty, notice this must be true for each induced copy of $C_4$
in $G$, otherwise, one could find such a copy of $C_4$ along with its nonempty
3-bag and proceed as in the other case. The following lemma uses this fact.

\begin{lemma}

    copy of $C_4$ having $Z$ as vertex set and free of the graphs depicted in
    Figures \ref{fig:chordal-p5-free-obstructions},
    \ref{fig:p5-house-obstruction-with-c5} and
    \ref{fig:p5-house-obstruction-no-c5}.
    \begin{lemenum}
        \item If $N_X$ is a 2-bag, then $N_X$ is stable. \label{lem:type_16}
        \item If $N_X$ is a 2-bag and $N_Y$ is a 1-bag with $X\cap Y
            =\varnothing$, then non-isolated vertices in $N_Y$ are anticomplete
            to $N_X$. \label{lem:type_14}
        \item Let $N_X$ be a 2-bag and let $E_X$ be the set of vertices in
            $N_\varnothing$ not anticomplete to $N_X$. If $E_X$ is nonempty,
            then there is a vertex $x$ in $N_X$ complete to $E_X$. Moreover, all
            vertices in $N_X\setminus \l{x}$ are anticomplete to all non-trivial
            connected components of $G[E_X]$. \label{lem:type_17}
        \item Let $N_X$ be a 2-bag and let $N_Y$ be a 1-bag such that $X\cap Y =
            \varnothing$. If $E_X$ is nonempty, then $N_Y$ is stable.
            \label{lem:type_18}
        \item Let $N_X$ and $N_Y$ be 2-bags. If $x\in N_X$ is not anticomplete
            to $N_\varnothing$, then $x$ is complete to $N_Y$.

    \end{lemenum} 
\end{lemma}

At last, we prove the main theorem.

\begin{theorem}
    A connected $(P_5,\text{house},C_5)$-free graph $G$ with an induced copy of
    $C_4$ is monopolar if and only if it does not contain any of the graphs
    depicted in Figures \ref{fig:chordal-p5-free-obstructions},
    \ref{fig:p5-house-obstruction-with-c5} and
    \ref{fig:p5-house-obstruction-no-c5}, as an induced subgraph.
\end{theorem}

\begin{proof}
    Since monopolarity is hereditary, it follows from
    \Cref{prop:minimal_obs_no_c5,prop:minimal_obs_c5,prop:minimal_obs_chordal}
    that $G$ does not contain the mentioned graphs as induced subgraphs.

    Due to \Cref{lem:type_1,lem:type_2}, the only possibly nonempty exact
    neighborhoods of the induced $4$-cycle $C$ are
    \[
        N_\varnothing, N_{a}, N_{b}, N_{c}, N_{d}, N_{ac}, N_{bd}, N_{abc}, N_{bcd},
        N_{acd} \text{ and } N_{abd}.
    \]
    We proceed by considering the following cases:

    \paragraph{Case 1: At least one 3-bag is nonempty.} Suppose without loss of
    generality that $N_{abc}$ is nonempty and let $b'\in N_{abc}$. By
    \Cref{lem:type_5}, both $N_{bcd}$ and $N_{abd}$ are empty, so the only other
    possibly nonempty 3-bag is $N_{acd}$. Furthermore, by \Cref{lem:type_6},
    suppose without loss of generality that $N_c$ is empty. Additionally,
    \Cref{lem:anti_1} ensures that $N_{abc}$ is anticomplete to $N_{acd}$, while
    \Cref{lem:type_3} implies that both sets are cliques. Therefore,
    \[
        N_{abc} \cup N_{acd} \cup \l{b,d}
    \]
    is a 2-cluster. Moreover, \Cref{lem:type_3bag_2} implies that $G[N_a]$ is a
    cluster, and \Cref{lem:anti_3} ensures that $N_a$ is anticomplete to both
    $N_{abc}$ and $N_{acd}$, so
    \[
        N_{abc}\cup N_{acd} \cup N_a \cup \l{b,d}
    \]
    is a cluster.

    On the one hand, from \Cref{lem:type_3bag_1}, we have that $N_{bd}$ is
    stable, and from \Cref{lem:comp_1}, we have that $N_{bd}$ is complete to
    both $N_{abc}$ and $N_{acd}$. On the other hand, by \Cref{lem:anti_2},
    $N_{ac}$ is anticomplete to $N_a$, while \Cref{lem:type_4} implies that
    $G[N_{ac}]$ is a cluster. Also, $N_{abc}$ and $N_{acd}$ are anticomplete to
    $N_{ac}$ by \Cref{lem:anti_1}. Then, up to this point,
    \[
        \p{N_{bd} \cup \l{a,c} , N_{abc} \cup N_{acd} \cup N_{ac} \cup N_{a} 
        \cup \l{b,d}}
    \]
    is a partial monopolar partition of $G$.

    For $N_\varnothing$, let $E_{bd}$ be the set of vertices in
    $N_{\varnothing}$ that are not anticomplete to $N_{bd}$. By
    \Cref{lem:type_3bag_3}, $E_{bd}$ induces a cluster, while \Cref{lem:type_8}
    implies that such a set is anticomplete to all other bags but $N_{bd}$.
    Therefore,
    \[
        N_{abc} \cup N_{acd} \cup N_{ac} \cup N_{a} 
        \cup E_{bd} \cup \l{b,d}
    \]
    is a cluster. For vertices in $N_\varnothing\setminus E_{bd}$, by
    \Cref{lem:type_9}, they are anticomplete to all but one bag between
    $N_{abc}$, $N_{acd}$ and $N_{ac}$, and \Cref{lem:type_8} implies that
    $N_\varnothing\setminus E_{bd}$ is anticomplete to $E_{bd}$. Consider the
    following subcases:

    \paragraph{Case 1.1: $N_\varnothing\setminus E_{bd}$ is not anticomplete to
    $N_{ac}$.} Let $S$ be the set of vertices in trivial components of
    $G[N_\varnothing\setminus E_{bd}]$, and let $\mathcal{C}$ be the set of
    connected components of $G[N_\varnothing\setminus (E_{bd} \cup S)]$. Notice
    $N_\varnothing \setminus E_{bd} \subseteq E_{ac}$ (in fact, $N_\varnothing
    \setminus E_{bd} = E_{ac}$). Thus, by the proof of \Cref{lem:type_3bag_4},
    if $C$ is the unique nontrivial component of $G[E_{ac}]$ (if it exists), we
    have that
    \[
        N_{abc} \cup N_{acd} \cup N_{ac} \cup N_{a} 
        \cup E_{bd} \cup K_C \cup \l{b,d}
    \]
    is a cluster, while
    \[
        N_{bd} \cup S \cup S_C \cup \l{a,c}
    \]
    is a stable set, where $(S_C,K_C)$ is a split partition of $C$.
    
    Finally, \Cref{lem:type_6} ensures that one between $N_{b}$ and $N_{d}$ is
    empty. Take the following subcases:

    \paragraph{Case 1.1.1: $N_d$ is empty.} Due to \Cref{lem:anti_2}, $N_b$ is
    anticomplete to $N_{bd}$. Let $K(N_b,N_{abc})$ be the set of vertices in
    $N_b$ that are complete to $N_{abc}$, and let $D = N_b \setminus
    K(N_b,N_{abc})$. By \Cref{lem:type_11}, $D$ is stable, so
    \[
        N_{bd} \cup D \cup S \cup S_C \cup \l{a,c}
    \]
    is a stable set. Additionally, \Cref{lem:type_11} implies that
    $K(N_b,N_{abc})$ is anticomplete to $D$.
    
    Let $F$ be the set of vertices in $K(N_b,N_{abc})$ that are not anticomplete
    to the set $N_{ac}$. By \Cref{lem:type_3bag_6}, $F$ is stable, and the same
    lemma implies that $F$ is anticomplete to $K(N_b,N_{abc})\setminus F$. Since
    $K(N_b,N_{abc})$ is anticomplete to $D$, $F$ is anticomplete to $D$, so
    \[
        N_{bd} \cup D \cup F \cup S \cup S_C \cup \l{a,c}
    \]
    is a stable set. Finally, notice that $G[K(N_b,N_{abc})\setminus F]$ is
    split by \Cref{lem:type_3bag_51}. Then, there is a partition of
    $K(N_b,N_{abc})\setminus F$ into $S_b$ a stable set and $K_b$ a clique.

    F$ is anticomplete to $F$, it follows that $S_b$ is anticomplete to both $D$
    and $F$. Meanwhile, we already know that $K_b$ is complete to $N_{abc}$,
    anticomplete to $N_{ac}$, and it is anticomplete to $N_{acd}$ and $N_a$ due
    to \Cref{lem:anti_4,lem:anti_3}, respectively. It follows that
    \begin{align*}

            & N_{abc} \cup N_{acd} \cup N_{ac} \cup N_{a} \cup K_b \cup E_{bd} 
            \cup K_C \cup \l{b,d} \bigg)
    \end{align*}

    \paragraph{Case 2.2: $N_\varnothing$ is nonempty.} Let $E_{ac}$ be the set
    of vertices in $N_\varnothing$ that are not anticomplete to $N_{ac}$.
    Similarly, let $E_{bd}$ be the set of vertices in $N_\varnothing$ that are
    not anticomplete to $N_{bd}$. Since all 3-bags are empty, and due to
    \Cref{lem:anti_5}, we have that $N_\varnothing = E_{ac} \cup E_{bd}$.
    Moreover, \Cref{lem:type_8} implies that $E_{ac}$ and $E_{bd}$ partition
    $N_{\varnothing}$, with $E_{ac}$ anticomplete to $E_{bd}$. Suppose without
    loss of generality that $E_{ac}\neq \varnothing$.

    It follows from \Cref{lem:type_18} that $N_b$ is stable. Meanwhile,
    \Cref{lem:type_17} implies that there is a vertex $x\in N_{ac}$ complete to $E_{ac}$,
    while $N_{ac}\setminus \l{x}$ is anticomplete to all nontrivial connected
    components of $G[E_{ac}]$.
    
    Observe that $\p{a,b,c,x,a}$ is an induced copy of $C_4$ with the set
    $E_{ac}$ complete to $x$ and anticomplete to $\l{a,b,c}$. Then, by
    \Cref{lem:type_7}, $G[E_{ac}]$ is either a cluster or split. Consider the
    following cases:

    \paragraph{Case 2.2.1: $G[E_{ac}]$ is not a cluster.} Then $G[E_{ac}]$ is
    split, so let $(S,K)$ be a split partition of $E_{ac}$ with $K$ a maximal
    nontrivial clique. Additionally, since $\p{a,b,c,x,a}$ is an induced copy of
    $C_4$ and, by \Cref{lem:anti_5}, $N_{a}$ is anticomplete to $E_{ac}$, it
    follows from \Cref{lem:type_12} that $G[N_a]$ is a cluster. Recall that
    \Cref{lem:anti_2} implies that $N_{ac}$ is a anticomplete to $N_a$, while
    \Cref{lem:anti_5} implies that $N_{b}$ is anticomplete to $K$. If $K$ is
    empty, then it is obvious that
    \[
        N_{ac} \cup N_a  \cup K \cup \l{b,d}
    \]
    is a cluster. If $K$ is nonempty, then it is nontrivial, so it must be the
    case that $N_{ac} \setminus \l{x}$ is anticomplete to $K$ as per
    \Cref{lem:type_17}. At any rate, it follows that
    \[
        N_{ac} \cup N_a \cup K \cup \l{b,d}
    \]
    is a cluster.

    Meanwhile, recall that \Cref{lem:anti_2} implies that $N_{bd}$ is a
    anticomplete to $N_b$. Since $S\subseteq E_{ac}$ and \Cref{lem:anti_5}
    implies that $N_{b}$ is anticomplete $E_{ac}$, we have that
    \[
        N_{bd} \cup N_b \cup S \cup \l{a,c}
    \]
    is a stable set. If $E_{bd}$ is empty, then we are done and
    \[
        \p{N_{bd} \cup N_b \cup S  \cup \l{a,c}, N_{ac} \cup N_a \cup K
        \cup \l{b,d}}
    \]
    is a monopolar partition of $G$. If $E_{bd}$ is nonempty, then by
    \Cref{lem:type_18} we have that $N_a$ is stable. Additionally,
    \Cref{lem:type_17} implies that there is a $x'\in N_{bd}$ complete to
    $E_{bd}$. Moreover, by \Cref{lem:type_19}, $x$ is adjacent to $x'$.

    Notice that $\p{x,c,d,x',x}$ is an induced copy of $C_4$ with $E_{ac}$
    anticomplete to $\l{c,d,x'}$ and complete to $x$, $E_{bd}$ is anticomplete
    to $\l{c,d,x}$ and complete to $x'$, and $E_{ac}$ is anticomplete to
    $E_{bd}$. Then, by \Cref{lem:type_12}, at least one between $G[E_{ac}]$ and
    $G[E_{bd}]$ is a cluster. Since $G[E_{ac}]$ is not a cluster by hypothesis,
    we have that $G[E_{bd}]$ is a cluster anticomplete to $N_a$, $N_{ac}$ and
    $E_{ac}$, so
    \[
        \p{N_{bd} \cup N_b \cup S  \cup \l{a,c}, N_{ac} \cup N_a \cup K \cup E_{bd}
        \cup \l{b,d}}
    \]
    is a monopolar partition of $G$ (see
    \Cref{fig:monopolar_partition_c4_case_2_2_1}).

    \paragraph{Case 2.2.2: $G[E_{ac}]$ is a cluster.} We can consider
    $G[E_{bd}]$ to be a cluster as well, since one could proceed as in the
    previous case if $G[E_{bd}]$ is not a cluster. Additionally, if $G[E_{ac}]$
    happens to be split, then once again we can proceed as in the previous case,
    so consider $G[E_{ac}]$ to be not split.

    Similar to the argument that at least one of $G[E_{ac}]$ and $G[E_{bd}]$ is
    a cluster, by \Cref{lem:type_13} we have that $G[E_{bd}]$ is split. As in
    the beginning of the previous case, $\p{a,b,c,x,a}$ is an induced copy of
    $C_4$, with $E_{ac}$ anticomplete to $\l{a,b,c}\cup N_{a}$ and complete to
    $x$. Then, by \Cref{lem:type_13}, it follows that $G[N_a]$ is split. Let
    $(S,K)$ be a split partition of $N_a$ where $K$ is a maximal nontrivial
    clique.

    Notice $S$ is anticomplete to $N_{ac}$ by \Cref{lem:anti_2}, so
    \[
        N_{ac} \cup S \cup \l{b,d}
    \]
    is stable. Meanwhile, recall that $N_b$ is stable, so it induces a cluster.
    Moreover, \Cref{lem:anti_2} ensures that $N_{bd}$ is anticomplete to $N_b$,
    and \Cref{lem:anti_5} implies that $E_{ac}$ is anticomplete to both $N_b$
    and $K$. Therefore,
    \[
        N_{bd} \cup N_b \cup K \cup E_{ac} \cup \l{a,c}
    \]
    is a cluster, noting that $K\cup \l{a}$ is a clique. If $E_{bd}$ is stable,
    then it is clear that
    \[
        \p{N_{ac} \cup S \cup E_{bd} \cup \l{b,d} , N_{bd} \cup N_b \cup K 
        \cup E_{ac} \cup \l{a,c}}
    \]
    is a monopolar partition of $G$. If $E_{bd}$ is not stable, in particular is
    not empty, so \Cref{lem:type_18} implies that $N_a$ is stable, so $S=N_a$
    and $K=\varnothing$. Since $E_{bd}$ induces a graph that is both split and a
    cluster, it consists of a disjoint union of a clique and a stable set. Let
    $(S',K')$ be a split partition of $G[E_{bd}]$ such that $S'$ is anticomplete
    to $K'$, with $K'$ a maximal nontrivial clique. As in the previous case,
    recall that, if $K'$ is nonempty, then it is complete to a unique $x'\in
    N_{bd}$ and it is anticomplete to $N_{bd}\setminus \l{x'}$. It follows that
    \[
        N_{ac} \cup N_a \cup S' \cup \l{b,d}
    \]
    is a stable set, while
    \[
        N_{bd} \cup N_b \cup E_{ac} \cup K' \cup \l{a,c}
    \]
    is a cluster, so
    \[
        \p{N_{ac} \cup N_a \cup S' \cup \l{b,d} , N_{bd} \cup N_b \cup
        E_{ac} \cup K' \cup \l{a,c}}
    \]
    is a monopolar partition of $G$ (see
    \Cref{fig:monopolar_partition_c4_case_2_2_2}).
