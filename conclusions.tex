\chapter{Conclusions}

\begin{corollary}
    There is a linear-time certyfing algorithm that decides whether a
    $(P_5,\text{house})$-free graph is monopolar.
\end{corollary}

\begin{proposition}
    The problem of decing whether a graph has a $(k,\infty)$-polar partition for
    $k\geq 1$ is NP-complete.
\end{proposition}
